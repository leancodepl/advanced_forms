One subform per step is what makes per-step validation one line: `await step.validate()` reaches that step's fields and
nothing else, so a field on a later step is never flagged before the user gets there. The final submit calls the
parent's `validate()`, which walks every attached step, so a step the user walked back over is checked again.

```dart title="wizard_steps.dart"
/// One step of the wizard: an ordinary form controller with a title.
abstract class WizardStep extends AdvancedFormController {
  String get title;
}

class AccountStep extends WizardStep {
  AccountStep() {
    registerFields([email, password]);
  }

  @override
  String get title => 'Account';

  final email = AdvancedTextFieldController(
    validator: (value) => value.contains('@') ? null : 'Enter an email',
  );
  final password = AdvancedTextFieldController(
    validator: atLeastLength(8, 'At least 8 characters'),
  );
}

class AddressStep extends WizardStep {
  AddressStep() {
    registerFields([country, city]);
    // Picking another country clears the city — a relation, not a rule.
    addRelation(country, (value) => value, (_) => city.reset());
  }

  @override
  String get title => 'Address';

  final country = AdvancedSingleSelectFieldController<String, String>(
    initialValue: null,
    options: const ['Poland', 'Germany', 'Spain'],
    validator: notNull('Pick a country'),
  );
  final city = AdvancedTextFieldController(validator: filled('Enter a city'));
}

class ConfirmStep extends WizardStep {
  ConfirmStep() {
    registerFields([terms]);
  }

  @override
  String get title => 'Confirm';

  final terms = AdvancedBooleanFieldController<String>(
    validator: mustBeTrue('You have to accept the terms'),
  );
}
```

```dart title="wizard_controller.dart"
/// The wizard owns no fields of its own — every field lives on a step.
/// Which step is on screen is navigation, not form state, so it lives here
/// as a plain field with notifyListeners().
class WizardController extends AdvancedFormController {
  WizardController() {
    steps.forEach(addSubform);
  }

  final account = AccountStep();
  final address = AddressStep();
  final confirm = ConfirmStep();

  late final steps = <WizardStep>[account, address, confirm];

  int get index => _index;
  var _index = 0;

  WizardStep get current => steps[_index];
  bool get isFirst => _index == 0;
  bool get isLast => _index == steps.length - 1;

  /// Validates the current step only, and moves on if it passed.
  Future<bool> next() async {
    final step = current;
    if (!await step.validate()) {
      // Shown its errors once — let it correct itself as the user types. A
      // step's own mode wins over the parent's, so later steps stay quiet.
      step.setValidationMode(ValidationMode.onUserInteraction);
      return false;
    }
    _goTo(_index + 1);
    return true;
  }

  void back() => _goTo(_index - 1);

  /// The parent's validate() reaches every step, including ones already
  /// passed — a value edited on the way back is checked again.
  Future<bool> submit() async {
    if (await validate()) {
      return true;
    }
    // validate() just ran, so canSubmit is an accurate snapshot here.
    final broken = steps.indexWhere((step) => !step.value.canSubmit);
    if (broken != -1) {
      steps[broken].setValidationMode(ValidationMode.onUserInteraction);
      _goTo(broken);
    }
    return false;
  }

  void _goTo(int index) {
    if (index >= 0 && index < steps.length && index != _index) {
      _index = index;
      notifyListeners();
    }
  }
}
```

```dart title="wizard_form.dart"
class WizardForm extends StatefulWidget {
  const WizardForm({super.key});

  @override
  State<WizardForm> createState() => _WizardFormState();
}

class _WizardFormState extends State<WizardForm> {
  final _wizard = WizardController();

  @override
  void dispose() {
    _wizard.dispose();
    super.dispose();
  }

  Future<void> _submit() async {
    final done = await _wizard.submit();
    if (mounted) {
      final message = done ? 'Account created.' : 'Some steps need fixing.';
      ExampleLog.of(context).add(message);
    }
  }

  @override
  Widget build(BuildContext context) {
    return ListenableBuilder(
      listenable: _wizard,
      builder: (context, _) {
        final step = _wizard.current;
        return Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Text(
              'Step ${_wizard.index + 1} of ${_wizard.steps.length} — '
              '${step.title}',
              style: Theme.of(context).textTheme.titleMedium,
            ),
            const SizedBox(height: 8),
            LinearProgressIndicator(
              value: (_wizard.index + 1) / _wizard.steps.length,
            ),
            const SizedBox(height: 12),
            switch (step) {
              final AccountStep step => Column(
                children: [
                  DocsTextField(field: step.email, label: 'Email'),
                  DocsTextField(
                    field: step.password,
                    label: 'Password',
                    obscureText: true,
                  ),
                ],
              ),
              final AddressStep step => Column(
                children: [
                  DocsDropdownField(field: step.country, label: 'Country'),
                  DocsTextField(field: step.city, label: 'City'),
                ],
              ),
              final ConfirmStep step => DocsSwitchField(
                field: step.terms,
                label: 'I accept the terms',
              ),
              _ => const SizedBox.shrink(),
            },
            DocsActions(
              children: [
                if (!_wizard.isFirst)
                  TextButton(
                    onPressed: _wizard.back,
                    child: const Text('Back'),
                  ),
                if (_wizard.isLast)
                  DocsSubmitButton(
                    form: _wizard,
                    label: 'Create account',
                    onPressed: _submit,
                  )
                else
                  DocsSubmitButton(
                    form: step,
                    label: 'Next',
                    onPressed: _wizard.next,
                  ),
              ],
            ),
          ],
        );
      },
    );
  }
}
```

Three decisions in that code are worth keeping:

- **Create the wizard controller above the pages.** A controller built inside a step page dies with it, taking that
  step's values and errors along. With routes, provide it above the `Navigator` or hand it to each route with
  `ChangeNotifierProvider.value`.
- **Escalate the failed step, not the whole form.** After a failed `next()`, the step gets
  `ValidationMode.onUserInteraction` so it corrects itself as the user types. Because a subform's own mode wins over the
  parent's, the steps ahead stay in `manual` and never flag an untouched field.
- **A conditional step stays attached with `setValidationEnabled(false)`.** The example app's *Step Form* screen has an
  invoice step that a switch on the address step adds or drops through `addRelation`. Switched off, the step leaves
  the flow *and* `validate()`, `canSubmit` and `validating`, so one flag is the whole condition — and its values survive
  in case the user flips the switch back.
